```moonbit
  implementations: List[Implementation]

  name: String
  interface: String

  name: String
  interface: String
  wit_file: String

  name: String
  type: String
  path: String

}

struct AppConfig {
  api_version: String
  kind: String
  metadata: Metadata
  spec: Spec
}

struct Metadata {
  name: String
  version: String
}

struct Spec {
  components: List[ComponentConfig]
}
```

### 任务 2: Wasm 组件加载器（2-3 周）

**状态：未开始**
**优先级：⭐⭐⭐ 高

#### 2.1 选择 Wasm 运行时

**选项评估**
| 运行时 | 优势 | 劣势 |
|--------|------|
| Wasmtime | 成熟、WASI 支持好 | 可能需要 FFI |
| WAMR | 轻量、适合嵌入式 | 功能相对简单 |
| Wasm3 | 极快、最小 | 功能有限 |

**建议**：先用 Wasmtime，有丰富文档和社区支持

#### 2.2 组件加载器接口

**文件** `src/component/loader.mbt`

```moonbit
// 组件实例
struct ComponentInstance {
  id: Int
  name: String
  wasm_bytes: List[Byte]?
  // 运行时内部数据
}

// 加载组件
fn load_wasm_component(path: String) -> Result[ComponentInstance, Error]

// 调用组件函数
fn call_component(
  comp: ComponentInstance, 
  func_name: String, 
  args: List[Value]
) -> Result[Value, Error]

// 销毁组件
fn destroy_component(comp: ComponentInstance) -> Unit
```
