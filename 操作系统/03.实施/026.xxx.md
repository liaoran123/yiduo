# 当前代码架构分析与重构建议

## 核心洞察：「管位置」 vs「管血液」

在讨论重构之前，先明确一多 OS 与传统编排系统的本质差异：

| 维度 | Nomad / K8s (传统编排) | 一多 OS (可编程调度) |
|------|-----------------------|---------------------|
| **调度定位** | **管位置** - 集群资源分配器 | **管血液** - 应用级拓扑与生命周期管理者 |
| **调度粒度** | 粗粒度（Pod / Task / 容器级别） | 细粒度（Wasm 组件 / 接口调用级别） |
| **交互管理** | 不管（交由容器内的应用自行处理） | **强接管**（基于 WIT 契约自动路由、降级） |
| **通讯方式** | 依赖网络栈（HTTP/TCP）或系统 IPC | 零拷贝共享内存 + 链式数据自然流动 |

**因为一多 OS 要「管血液」，所以重构方向必须是：**
- 🔴 **内联核心**：管血液流动的「血管系统」（消息路由、零拷贝、缓存）
- 🟢 **Wasm组件**：血液中流动的「细胞」（能力提供者、业务逻辑）

/// 一多 OS 多层缓存架构
/// L1: 组件内联缓存（直接访问，<10ns）
/// L2: 节点级共享缓存（<100ns）
/// L3: 持久化缓存（<1ms）

/// L1缓存：组件内联，直接函数指针
struct L1Cache {
  /// 能力绑定缓存：CapabilityId -> 直接函数指针
  capability_bindings: FixedArray[Option[CapabilityFunction]],
  /// 常用组件引用
  hot_components: FixedArray[Option[HotComponentRef]]
}

/// L2缓存：节点级共享
struct L2Cache {
  /// 能力索引：InterfaceId -> 可用实现列表
  capability_index: HashMap[InterfaceId, List[CapabilityMeta]],
  /// 已加载能力提供者
  loaded_providers: HashMap[CapabilityId, CapabilityProvider]
}

/// L3缓存：持久化
struct L3Cache {
  /// 组件索引数据库
  component_index: KVStore,
  /// 能力元数据存储
  capability_meta: KVStore
}

/// 缓存统一访问接口
struct CacheSystem {
  l1: L1Cache,
  l2: L2Cache,
  l3: L3Cache
}

/// 快速查找能力（优先L1，然后L2，最后L3）
fn find_capability(cache: CacheSystem, query: CapabilityQuery) -> Option<CapabilityInstance> {
  // L1: 直接函数指针查找（<10ns）
  if let Some(cap) = cache.l1.get(query.capability_id) {
    return Some(cap)
  }
  
  // L2: 索引查找（<100ns）
  if let Some(cap) = cache.l2.find(query) {
    // 回填L1
    cache.l1.populate(query.capability_id, cap)
    return Some(cap)
  }
  
  // L3: 持久化查找（<1ms）
  if let Some(cap) = cache.l3.find(query) {
    // 回填L2和L1
    cache.l2.populate(query.capability_id, cap)
    cache.l1.populate(query.capability_id, cap)
    return Some(cap)
  }
  
  None
}
```

### 3. 新增能力提供者框架

/// 一多 OS 能力提供者框架（内联核心，Wasm实现）

/// 能力接口定义（WIT）
struct CapabilityInterface {
  name: String,
  version: String,
  functions: List[FunctionSignature]
}

/// 能力提供者实例
struct CapabilityProvider {
  id: CapabilityId,
  interface: CapabilityInterface,
  implementation_type: ImplementationType, // Native | Wasm | AOT
  /// Native实现：直接函数指针
  native_functions: Option<NativeFunctionTable>,
  /// Wasm实现：Wasm组件引用
  wasm_component: Option<WasmComponentRef>,
  /// 健康检查
  health_check: Option<HealthCheck>
}

/// 能力提供者框架（内联核心）
struct CapabilityFramework {
  /// 已注册的能力接口
  registered_interfaces: HashMap[InterfaceId, CapabilityInterface],
  /// 可用的实现
  available_providers: HashMap[InterfaceId, List[CapabilityProvider]],
  /// 当前绑定
  current_bindings: HashMap[ComponentId, HashMap[InterfaceId, CapabilityProvider]]
}

/// 注册能力接口
fn register_interface(framework: CapabilityFramework, interface: CapabilityInterface) -> CapabilityFramework {
  let mut new_map = framework.registered_interfaces
  new_map[interface.id()] = interface
  CapabilityFramework{
    registered_interfaces: new_map,
    ..framework
  }
}

/// 注册能力提供者实现
fn register_provider(framework: CapabilityFramework, interface_id: InterfaceId, provider: CapabilityProvider) -> CapabilityFramework {
  let mut new_map = framework.available_providers
  let existing = new_map.get_or_default(interface_id, [])
  new_map[interface_id] = provider :: existing
  CapabilityFramework{
    available_providers: new_map,
    ..framework
  }
}

/// 为组件绑定能力（预编译阶段完成，避免运行时查找）
fn bind_capability(framework: CapabilityFramework, component: ComponentId, interface: InterfaceId, provider: CapabilityProvider) -> CapabilityFramework {
  let mut new_bindings = framework.current_bindings
  let component_bindings = new_bindings.get_or_default(component, HashMap::new())
  let mut updated_bindings = component_bindings
  updated_bindings[interface] = provider
  new_bindings[component] = updated_bindings
  CapabilityFramework{
    current_bindings: new_bindings,
    ..framework
  }
}

### 4. 调度器增强：新增调度算法

/// Bin-Packing装箱算法（性能关键，必须内联）
/// 目标：最大化资源利用率

/// 节点资源状态
struct NodeResources {
  cpu_used: Int,
  cpu_total: Int,
  memory_used: Int,
  memory_total: Int,
  gpu_used: Int,
  gpu_total: Int
}

/// 调度请求
struct SchedulingRequest {
  component: ComponentId,
  cpu_needed: Int,
  memory_needed: Int,
  gpu_needed: Int,
  affinity_rules: List[AffinityRule],
  anti_affinity_rules: List[AntiAffinityRule]
}

/// 装箱调度核心（<1μs完成）
fn best_fit_packing(nodes: List[NodeResources], request: SchedulingRequest) -> Option[NodeId] {
  let mut best_node: Option[NodeId] = None
  let mut best_fragmentation = Int::MAX
  
  for node in nodes {
    // 检查资源是否足够
    if can_fit(node, request) {
      // 检查亲和/反亲和规则
      if satisfies_affinity(node, request) && satisfies_anti_affinity(node, request) {
        // 计算碎片化程度
        let fragmentation = calculate_fragmentation(node, request)
        if fragmentation < best_fragmentation {
          best_fragmentation = fragmentation
          best_node = Some(node.id)
        }
      }
    }
  }
  
  best_node
}

/// 第一次适应算法
fn first_fit_packing(nodes: List[NodeResources], request: SchedulingRequest) -> Option[NodeId] {
  for node in nodes {
    if can_fit(node, request) && 
       satisfies_affinity(node, request) && 
       satisfies_anti_affinity(node, request) {
      return Some(node.id)
    }
  }
  None
}
```
